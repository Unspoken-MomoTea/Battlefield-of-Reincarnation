<script>
(function () {
  var selectedYear = null;
  var selectedLabel = null;
  var selectedSwipeId = null;
  var worldbookSynced = false;
  var worldBook = null;

  /* 选择学年篇章 */
  window.anhsSelectYear = function (el) {
    if (el.hasAttribute('disabled')) return;

    var cards = document.querySelectorAll('.anhs-card');
    for (var i = 0; i < cards.length; i++) { cards[i].classList.remove('selected'); }
    el.classList.add('selected');

    selectedYear = el.getAttribute('data-year');
    selectedLabel = el.getAttribute('data-label');
    selectedSwipeId = Number(el.getAttribute('data-swipe'));
    worldbookSynced = false;

    var enterBtn = document.getElementById('anhsEnterBtn');
    enterBtn.disabled = true;
    enterBtn.textContent = '确认进入';

    var syncBtn = document.getElementById('anhsSyncBtn');
    syncBtn.disabled = false;

    document.getElementById('anhsSyncStatus').textContent = '已选择：' + selectedLabel + '，请先同步世界书';
  };

  /* 选择DLC入口 */
  window.anhsSelectDlc = function (el) {
    if (el.hasAttribute('disabled')) return;

    var cards = document.querySelectorAll('.anhs-card');
    for (var i = 0; i < cards.length; i++) { cards[i].classList.remove('selected'); }
    el.classList.add('selected');

    selectedYear = null;
    selectedLabel = el.getAttribute('data-label');
    selectedSwipeId = Number(el.getAttribute('data-swipe'));
    worldbookSynced = true;

    var enterBtn = document.getElementById('anhsEnterBtn');
    enterBtn.disabled = false;
    enterBtn.textContent = '进入终端';

    var syncBtn = document.getElementById('anhsSyncBtn');
    syncBtn.disabled = true;

    document.getElementById('anhsSyncStatus').textContent = '已选择：' + selectedLabel + '，可直接进入';
  };

  /* 同步世界书与初始化状态 */
  window.anhsSyncWorldbook = async function () {
    if (!selectedYear) {
      anhsToast('请先选择篇章', 'warning');
      return;
    }

    var btn = document.getElementById('anhsSyncBtn');
    var status = document.getElementById('anhsSyncStatus');

    btn.disabled = true;
    btn.textContent = '同步中...';
    status.textContent = '正在同步：' + selectedLabel;

    try {
      await anhsApplyYearWorldbook(selectedYear);
      worldbookSynced = true;
      document.getElementById('anhsEnterBtn').disabled = false;
      status.textContent = '已同步：' + selectedLabel + ' 且已初始化状态';
      anhsToast(selectedLabel + '世界书已同步', 'success');
    } catch (error) {
      console.error('[ANHS] 世界书同步失败:', error);
      status.textContent = '同步失败，请检查世界书API或手动切换';
      anhsToast('世界书同步失败', 'error');
    } finally {
      btn.disabled = false;
      btn.textContent = '同步并初始化';
    }
  };

  /* 确认进入 */
  window.anhsConfirmEnter = function () {
    if (!selectedLabel) {
      anhsToast('请先选择项目', 'warning');
      return;
    }
    if (!worldbookSynced) {
      anhsToast('请先同步世界书', 'warning');
      return;
    }
    anhsSwitchToOpening(selectedSwipeId);
  };

  /* 初始化世界书 */
  async function anhsInitWorldbook() {
    if (!worldBook && typeof getCharWorldbookNames === 'function') {
      worldBook = getCharWorldbookNames('current').primary;
    }
    return worldBook;
  }

  /* 批量设置条目启用状态(按前缀或包含) */
  async function anhsSetEntriesByTag(tag, enabled) {
    await anhsInitWorldbook();
    if (!worldBook || typeof getWorldbook !== 'function' || typeof replaceWorldbook !== 'function') return false;

    var entries = await getWorldbook(worldBook);
    var updated = entries.map(function(entry) {
      if (entry.name && entry.name.includes(tag)) {
        var copy = {};
        for (var k in entry) { if (Object.prototype.hasOwnProperty.call(entry, k)) copy[k] = entry[k]; }
        copy.enabled = enabled;
        return copy;
      }
      return entry;
    });
    await replaceWorldbook(worldBook, updated);
    return true;
  }

  /* 新增：精确匹配名称修改世界书状态 */
  async function anhsSetEntriesByExactName(nameStr, enabled) {
    await anhsInitWorldbook();
    if (!worldBook || typeof getWorldbook !== 'function' || typeof replaceWorldbook !== 'function') return false;

    var entries = await getWorldbook(worldBook);
    var updated = entries.map(function(entry) {
      if (entry.name && entry.name === nameStr) {
        var copy = {};
        for (var k in entry) { if (Object.prototype.hasOwnProperty.call(entry, k)) copy[k] = entry[k]; }
        copy.enabled = enabled;
        return copy;
      }
      return entry;
    });
    await replaceWorldbook(worldBook, updated);
    return true;
  }

  /* 新增：切换解析模式逻辑 */
  window.anhsSetParseMode = async function(mode) {
    var btnMain = document.getElementById('anhsModeMain');
    var btnExtra = document.getElementById('anhsModeExtra');

    // UI样式切换
    if (mode === 'main') {
      btnMain.classList.add('active');
      btnExtra.classList.remove('active');
    } else {
      btnExtra.classList.add('active');
      btnMain.classList.remove('active');
    }

    try {
      if (mode === 'main') {
        await anhsSetEntriesByExactName('output_format (随AI输出开，主API)', true);
        await anhsSetEntriesByExactName('[mvu_update]output_format (使用额外模型更新变量开)', false);
        anhsToast('已开启主变量解析模式', 'success');
      } else {
        await anhsSetEntriesByExactName('[mvu_update]output_format (使用额外模型更新变量开)', true);
        await anhsSetEntriesByExactName('output_format (随AI输出开，主API)', false);
        anhsToast('已开启额外解析模式', 'success');
      }
    } catch (e) {
      console.error('[ANHS] 模式切换失败:', e);
      anhsToast('模式切换失败，请检查世界书连接', 'error');
    }
  };

  /* 应用学年世界书（仅控制对应世界书条目的启闭，去除冗余的MVU初始化） */
  async function anhsApplyYearWorldbook(year) {
    var yearTagMap = { y1: 'S1_', y2: 'S2_', y3: 'S3_' };

    await anhsSetEntriesByTag('S1_', false);
    await anhsSetEntriesByTag('S2_', false);
    await anhsSetEntriesByTag('S3_', false);
    await anhsSetEntriesByTag(yearTagMap[year], true);
  }

  /* 切换开场白 */
  async function anhsSwitchToOpening(swipeId) {
    var btn = document.getElementById('anhsEnterBtn');
    btn.disabled = true;
    btn.textContent = '正在进入...';

    try {
      var messages = await getChatMessages('0', { include_swipe: true });
      if (messages && messages[0] && messages[0].swipes && messages[0].swipes[swipeId]) {
        await setChatMessage(messages[0].swipes[swipeId], 0, {
          swipe_id: swipeId,
          refresh: 'display_and_render_current'
        });
        anhsToast('已进入' + selectedLabel, 'success');
      } else {
        throw new Error('开场白 ' + (swipeId + 1) + ' 未找到');
      }
    } catch (error) {
      console.error('[ANHS] 切换开场白失败:', error);
      anhsToast('进入失败：' + error.message, 'error');
      btn.disabled = false;
      btn.textContent = '确认进入';
    }
  }

  /* Toast通知 */
  function anhsToast(message, type) {
    if (typeof toastr !== 'undefined') {
      if (type === 'success') toastr.success(message);
      else if (type === 'error') toastr.error(message);
      else if (type === 'warning') toastr.warning(message);
      else toastr.info(message);
    } else {
      console.log('[' + (type || 'info') + '] ' + message);
    }
  }
})();
</script>
