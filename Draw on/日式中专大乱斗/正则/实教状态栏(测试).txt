    function formatTime(raw) {
      if (!raw || raw === 'N/A') return '--';
      var parts = raw.split('-');

    /* 📊 Canvas绘制OAA四维雷达蛛网图 📊 */
    function drawOaaRadar(canvasId, abilities) {
      var canvas = document.getElementById(canvasId);
      if (!canvas) return;
      var ctx = canvas.getContext('2d');
      var width = canvas.width = 180;
      var height = canvas.height = 180;
      var cx = width / 2;
      var cy = height / 2;
      var maxRadius = 70;

      /* 轴名定义：学力(12点方向), 身体能力(3点), 灵活思考(6点), 社会贡献(9点) */
      var axes = ['学力', '身体能力', '灵活思考力', '社会贡献性'];
      var scores = axes.map(function(name) {
        return _.get(abilities, name + '.分数', 0);
      });

      ctx.clearRect(0, 0, width, height);

      /* 1. 绘制底色雷达网格(5圈蛛网) */
      ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--border').trim() || 'rgba(0, 243, 255, 0.2)';
      ctx.lineWidth = 0.5;
      for (var r = 1; r <= 5; r++) {
        var radius = maxRadius * (r / 5);
        ctx.beginPath();
        ctx.moveTo(cx, cy - radius);
        ctx.lineTo(cx + radius, cy);
        ctx.lineTo(cx, cy + radius);
        ctx.lineTo(cx - radius, cy);
        ctx.closePath();
        ctx.stroke();
      }

      /* 2. 绘制十字交叉轴线 */
      ctx.beginPath();
      ctx.moveTo(cx, cy - maxRadius); ctx.lineTo(cx, cy + maxRadius);
      ctx.moveTo(cx - maxRadius, cy); ctx.lineTo(cx + maxRadius, cy);
      ctx.stroke();

      /* 3. 绘制雷达分数多边形 */
      var points = [
        { x: cx, y: cy - maxRadius * (scores[0] / 100) }, /* 学力 */
        { x: cx + maxRadius * (scores[1] / 100), y: cy }, /* 身体能力 */
        { x: cx, y: cy + maxRadius * (scores[2] / 100) }, /* 灵活思考力 */
        { x: cx - maxRadius * (scores[3] / 100), y: cy }  /* 社会贡献性 */
      ];

      var accentColor = getComputedStyle(document.documentElement).getPropertyValue('--accent').trim() || '#00f3ff';

      ctx.beginPath();
      ctx.moveTo(points[0].x, points[0].y);
      for (var i = 1; i < points.length; i++) {
        ctx.lineTo(points[i].x, points[i].y);
      }
      ctx.closePath();
      ctx.fillStyle = accentColor.replace('rgb', 'rgba').replace(')', ', 0.3)'); /* 半透明填充 */
      ctx.fill();
      ctx.strokeStyle = accentColor;
      ctx.lineWidth = 1.5;
      ctx.stroke();

      /* 4. 绘制各端点得分小圆圈 */
      ctx.fillStyle = '#fff';
      points.forEach(function(pt) {
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, 2.5, 0, Math.PI * 2);
        ctx.fill();
        ctx.stroke();
      });

      /* 5. 标上雷达图文本 */
      ctx.font = '10px sans-serif';
      ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--text-secondary').trim() || '#00f3ff';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText('学力', cx, cy - maxRadius - 10);
      ctx.fillText('身体', cx + maxRadius + 15, cy);
      ctx.fillText('思考', cx, cy + maxRadius + 10);
      ctx.fillText('社会', cx - maxRadius - 15, cy);
    }

      }
    }

    function closeNpcDetail() {
      $('#sys-detail-view').hide();
      $('#sys-list-view').show();
    }

    async function init() {
      await waitGlobalInitialized('Mvu');
      populateData();

      eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, function() {
        populateData();
      });

      var savedTheme = localStorage.getItem('cote_status_theme');
      if (savedTheme) {
        $('.s-container').attr('data-theme', savedTheme);
        $('.theme-dot').removeClass('active');
        $('.theme-dot[data-t="' + savedTheme + '"]').addClass('active');
      }

      $('.s-tab-btn').on('click', function() {
        var target = $(this).data('tab');
        $('.s-tab-btn').removeClass('active');
        $(this).addClass('active');
        $('.s-panel').removeClass('active');
        $('#panel-' + target).addClass('active');
        if (target === 'sys') { closeNpcDetail(); }
      });

      $('.theme-dot').on('click', function() {
        var t = $(this).data('t');
        $('.s-container').attr('data-theme', t);
        $('.theme-dot').removeClass('active');
        $(this).addClass('active');
        localStorage.setItem('cote_status_theme', t);
        /* 主题改变后重新渲染画布 */
        populateData();
      });

      $(document).on('click', '.s-filter-btn', function() {
        $('.s-filter-btn').removeClass('active');
        $(this).addClass('active');
        currentFilter = $(this).data('filter');
        renderNPCList();
      });

      $(document).on('click', '.s-npc-item', function() {
        var name = $(this).data('npc');
        var gender = $(this).data('gender');
        showNpcDetail(name, gender);
      });

      $(document).on('click', '.s-detail-back', function() {
        closeNpcDetail();
      });

      $(document).on('click', '.s-card-toggle', function() {
        var target = $(this).data('target');
        var body = $('#' + target);
        var self = $(this);

        body.slideToggle(200, function() {
          self.toggleClass('open', body.is(':visible'));
        });
      });

      $(document).on('click', '.s-effect-capsule', function(e) {
        e.stopPropagation();
        var drawerId = $(this).data('drawer');
        var drawer = $('#' + drawerId);

        $(this).toggleClass('active');
        drawer.slideToggle(150);
      });
    }

    $(errorCatched(init));
  </script>
