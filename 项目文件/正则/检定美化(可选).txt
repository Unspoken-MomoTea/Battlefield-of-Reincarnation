      <script>
        (function() {
          'use strict';
          const currentScript = document.currentScript;
          if (!currentScript) return;
          const wrapper = currentScript.closest('.dice-wrapper');
          if (!wrapper) return;

          // [改] 加入 THP / ATK 关键词
          const sectionIcons = {
            '行动': '🎯', '内容': '📋', '推演': '🎲', '伤害构成': '⚔️',
            '结算': '📊', '变化': '🔄', '附加状态': '🏷️', '最终状态': '✅',
            '备注': '💡', '观测情报': '🔍', '组件架构': '🧩', '实体档案': '👤',
            '总结': '📌'
          };
          const sectionTypes = {
            '推演': 'check', '结算': 'result', '变化': 'change',
            '备注': 'note', '实体档案': 'entity'
          };

          function beautifyContent(text) {
              const keywords = ["DC","HP","THP","EP","DEF","MDEF","AP","ATK","MATK","vs"];
              const specialChars = {
                  '[': 'bracket', ']': 'bracket', '`': 'code',
                  '✅': 'success', '❌': 'fail', '💥': 'damage', '💚': 'heal',
                  '➞': 'arrow', '→': 'arrow', '|': 'separator'
              };
              const lines = text.trim().split('\n');
              let finalHtml = '';

              lines.forEach((line, index) => {
                  let cleanedLine = line.trim().substring(line.trim().startsWith('>') ? 1 : 0).trim();
                  let indentLevel = Math.floor((line.match(/^(\s*)/) || ['',''])[1].length / 2);

                  // [改] 分区标题检测: "字段名:" 独占一行时, 渲染为带图标的标题
                  const secMatch = cleanedLine.match(/^([^：:]{1,10})[：:]\s*$/);
                  if (secMatch && sectionIcons[secMatch[1]]) {
                      const fname = secMatch[1];
                      const stype = sectionTypes[fname] || '';
                      finalHtml += `<div class="content-line section-header ${stype}">${sectionIcons[fname]} ${fname}</div>`;
                      return;
                  }

                  let lineHtml = '';
                  let inCode = false;

                  for (let i = 0; i < cleanedLine.length; i++) {
                      const char = cleanedLine[i];
                      if (char === '`') {
                          inCode = !inCode;
                          lineHtml += inCode ? '<span class="tk-code">' : '</span>';
                          continue;
                      }
                      if (inCode) {
                          lineHtml += char;
                          continue;
                      }
                      if (specialChars[char]) {
                          lineHtml += `<span class="tk-${specialChars[char]}">${char}</span>`;
                          continue;
                      }
                      // DC + 数字: 整体标红 (如 DC42, DC 12)
                      const dcMatch = cleanedLine.substring(i).match(/^DC\s*\d+/);
                      if (dcMatch) {
                          const dcStr = dcMatch[0];
                          lineHtml += `<span class="tk-dc">${dcStr}</span>`;
                          i += dcStr.length - 1;
                          continue;
                      }
                      let matched = false;
                      for (const kw of keywords) {
                          if (cleanedLine.substring(i, i + kw.length) === kw && /(^|\s|[^a-zA-Z])/.test(cleanedLine[i-1] || ' ') && /(\s|$|[^a-zA-Z])/.test(cleanedLine[i+kw.length] || ' ')) {
                              lineHtml += `<span class="tk-keyword">${kw}</span>`;
                              i += kw.length - 1;
                              matched = true;
                              break;
                          }
                      }
                      if (matched) continue;
                      
                      const numMatch = cleanedLine.substring(i).match(/^(\d+\/\d+|\d*[Dd]\d+(?:\s*[+-]\s*\d+)?|-?\d+%?)/);
                      if (numMatch) {
                          const numStr = numMatch[0];
                          const isDice = /[Dd]/.test(numStr);
                          lineHtml += `<span class="${isDice ? 'tk-dice' : 'tk-number'}">${numStr}</span>`;
                          i += numStr.length - 1;
                          continue;
                      }

                      lineHtml += char;
                  }
                  
                  let className = 'content-line';
                  if (index === 0) className += ' first-line';
                  else if (indentLevel > 0) className += ` indent-${Math.min(indentLevel, 3)}`;
                  finalHtml += `<div class="${className}">${lineHtml}</div>`;
              });
              return finalHtml;
          }

          const btn = wrapper.querySelector('.dice-btn');
          const rawDiv = wrapper.querySelector('.dice-raw');
          const labelSpan = wrapper.querySelector('.dice-label');
          const contentDiv = wrapper.querySelector('.dice-content');

          if (!btn || !rawDiv || !labelSpan || !contentDiv) return;

          const rawText = rawDiv.textContent;
          const firstLine = rawText.trim().split('\n')[0] || '检定结果';
          // [改] 去掉了 .replace(/^[^\w\s\u4e00-\u9fa5]+/, '') — 不再吞掉前导 [
          const cleanLabel = firstLine.replace(/^\s*>\s*/, '').trim();
          
          labelSpan.textContent = cleanLabel;
          contentDiv.innerHTML = beautifyContent(rawText);

          btn.addEventListener('click', function() {
            btn.classList.toggle('active');
            contentDiv.classList.toggle('show');
          });

        })();
      </script>
    </div>
    <!-- 替换模板结束 -->
