    function getMessageData() {
      var chatMessages = getChatMessages(getCurrentMessageId());
      if (!chatMessages || chatMessages.length === 0) return null;
      return chatMessages[0].message;
    }

    function extractActionInfo(messageText) {
      /* 找到所有 action_info 块 */
      var blocks = [];
      var regex = /<action_info>([\s\S]*?)<\/action_info>/g;
      var match;
      while ((match = regex.exec(messageText)) !== null) {
        blocks.push(match[1]);
      }
      return blocks;
    }

    function parseFieldLine(line) {
      /* 解析 | 字段: 值 | 格式 */
      var cleaned = line.replace(/^\|?\s*/, '').replace(/\s*\|?$/, '').trim();
      var colonIdx = cleaned.indexOf(':');
      if (colonIdx === -1) return null;
      var key = cleaned.substring(0, colonIdx).trim();
      var value = cleaned.substring(colonIdx + 1).trim();
      return { key: key, value: value };
    }

    function parseBlock(content) {
      var data = {
        title: '',
        type: '',
        skill: '',
        attacker: { name: '', stat: '', team: '', modifier: '' },
        defender: { name: '', stat: '', team: '', modifier: '' },
        diceAtk: '',
        diceDef: '',
        formula: '',
        result: '',
        exp: ''
      };

      var lines = content.split('\n');
      for (var i = 0; i < lines.length; i++) {
        var line = lines[i].trim();
        if (!line) continue;

        /* 标题行 {xxx} */
        if (line.match(/^\{.*\}$/)) {
          data.title = line.replace(/[{}]/g, '');
          continue;
        }

        var parsed = parseFieldLine(line);
        if (!parsed) continue;

        var key = parsed.key;
        var val = parsed.value;

        if (key === '类型') {
          data.type = val;
        } else if (key === '检定项') {
          data.skill = val;
        } else if (key.includes('攻方') || key.includes('主导')) {
          /* 解析攻方行：角色名 (属性值:X) | 团队修正: ... | 其他修正: ... */
          var atkParts = val.split('|');
          var nameMatch = (atkParts[0] || '').match(/^(.+?)\s*\(属性值[：:]?\s*(\d+)\)/);
          if (nameMatch) {
            data.attacker.name = nameMatch[1].trim();
            data.attacker.stat = nameMatch[2];
          } else {
            data.attacker.name = (atkParts[0] || '').trim();
          }
          for (var j = 1; j < atkParts.length; j++) {
            var part = atkParts[j].trim();
            if (part.includes('团队修正')) {
              data.attacker.team = part.replace(/团队修正[：:]\s*/, '');
            } else if (part.includes('其他修正')) {
              data.attacker.modifier = part.replace(/其他修正[：:]\s*/, '');
            }
          }
        } else if (key.includes('守方') || key.includes('目标')) {
          var defParts = val.split('|');
          var defMatch = (defParts[0] || '').match(/^(.+?)\s*\((属性值|DC)[/／]?(?:DC)?[：:]?\s*(\d+)\)/);
          if (defMatch) {
            data.defender.name = defMatch[1].trim();
            data.defender.stat = defMatch[3];
          } else {
            data.defender.name = (defParts[0] || '').trim();
          }
          for (var k = 1; k < defParts.length; k++) {
            var dpart = defParts[k].trim();
            if (dpart.includes('团队修正')) {
              data.defender.team = dpart.replace(/团队修正[：:]\s*/, '');
            } else if (dpart.includes('其他修正')) {
              data.defender.modifier = dpart.replace(/其他修正[：:]\s*/, '');
            }
          }
        } else if (key === '掷骰') {
          /* 攻方d100(72) vs 守方d100(45) */
          var diceMatch = val.match(/攻方d100\((\d+)\)\s*vs\s*守方d100\((\d+)/);
          if (diceMatch) {
            data.diceAtk = diceMatch[1];
            data.diceDef = diceMatch[2];
          } else {
            var singleDice = val.match(/d100\((\d+)\)/);
            if (singleDice) {
              data.diceAtk = singleDice[1];
            }
          }
        } else if (key === '判定公式') {
          data.formula = val;
        } else if (key === '结果') {
          data.result = val;
        } else if (key.includes('经验结算')) {
          data.exp = val;
        }
      }
      return data;
    }

    function getResultClass(result) {
      if (result.includes('大成功')) return 'res-crit';
      if (result.includes('大失败')) return 'res-fumble';
      if (result.includes('成功') || result.includes('反制')) return 'res-success';
      if (result.includes('失败')) return 'res-fail';
      if (result.includes('僵持') || result.includes('险胜')) return 'res-stalemate';
      if (result.includes('压制')) return 'res-dominate';
      return 'res-success';
    }

    function getDiceClass(val) {
      var n = parseInt(val, 10);
      if (n >= 96) return 'crit';
      if (n <= 5) return 'fumble';
      return '';
    }

    function isSoloChallenge(data) {
      return data.type.includes('单人') || data.defender.name.includes('DC') || !data.diceDef;
    }

    function renderPanel(data) {
      var solo = isSoloChallenge(data);
      var html = '';

      /* 顶部 */
      html += '<div class="ac-header">';

      html += '<div class="ac-side">';

      html += '<div class="ac-side-stat">';
      if (data.attacker.stat) html += '<span>属性值: ' + data.attacker.stat + '</span>';
      if (data.attacker.team) html += '<span>团队: ' + data.attacker.team + '</span>';
      if (data.attacker.modifier) html += '<span>修正: ' + data.attacker.modifier + '</span>';
      html += '</div></div>';

      if (!solo) {
        html += '<div class="ac-vs-icon">VS</div>';

        /* 守方 */
        html += '<div class="ac-side">';
        html += '<div class="ac-side-label">守方 / 目标</div>';
        html += '<div class="ac-side-name">' + (data.defender.name || '--') + '</div>';
        html += '<div class="ac-side-stat">';
        if (data.defender.stat) html += '<span>属性值/DC: ' + data.defender.stat + '</span>';
        if (data.defender.team) html += '<span>团队: ' + data.defender.team + '</span>';
        if (data.defender.modifier) html += '<span>修正: ' + data.defender.modifier + '</span>';
        html += '</div></div>';
      } else {
        /* 单人挑战显示DC */
        html += '<div class="ac-vs-icon">→</div>';
        html += '<div class="ac-side">';
        html += '<div class="ac-side-label">难度阈值 (DC)</div>';
        html += '<div class="ac-side-name">' + (data.defender.stat || data.defender.name || '--') + '</div>';
        html += '<div class="ac-side-stat"></div></div>';
      }
      html += '</div>';

      /* 掷骰 */
      if (data.diceAtk) {
        html += '<div class="ac-dice-row">';
        html += '<div class="ac-dice">';
        html += '<div class="ac-dice-label">d100</div>';
        html += '<div class="ac-dice-val ' + getDiceClass(data.diceAtk) + '">' + data.diceAtk + '</div>';
        html += '</div>';
        if (data.diceDef && !solo) {
          html += '<div class="ac-dice-vs">vs</div>';
          html += '<div class="ac-dice">';
          html += '<div class="ac-dice-label">d100</div>';
          html += '<div class="ac-dice-val ' + getDiceClass(data.diceDef) + '">' + data.diceDef + '</div>';
          html += '</div>';
        }
        html += '</div>';
      }

      /* 公式 */
      if (data.formula) {
        html += '<div class="ac-formula">' + data.formula + '</div>';
      }

      /* 结果 */
      if (data.result) {
        html += '<div class="ac-result ' + getResultClass(data.result) + '">' + data.result + '</div>';
      }

      /* 经验 */
      if (data.exp) {
        html += '<div class="ac-exp"><span class="ac-exp-icon">⬆</span>' + data.exp + '</div>';
      }

      return html;
    }

    /* 动态主题同步监听魔法：自动读取并同步父级面板的主题属性 */
    function syncTheme() {
      try {
        if (window.frameElement && window.parent) {
          /* 尝试读取主控制面板的 data-theme */
          var parentPanel = window.parent.document.querySelector('.cote-panel');
          if (parentPanel) {
            var theme = parentPanel.getAttribute('data-theme');
            if (theme) {
              document.documentElement.setAttribute('data-theme', theme);
            }
          }
        }
      } catch (e) {
        console.warn('[OAA检定] 主题自动同步受限于同源策略或暂无主面板:', e);
      }
    }

    function init() {
      syncTheme(); /* 执行主题同步 */
      try {
        var messageText = getMessageData();
        if (!messageText) {
          document.getElementById('ac-root').innerHTML = '<div class="ac-loading">❌ 无法获取消息</div>';
          return;
        }

        var blocks = extractActionInfo(messageText);
        if (blocks.length === 0) {
          document.getElementById('ac-root').innerHTML = '<div class="ac-loading">❌ 未找到检定数据</div>';
          return;
        }

        var currentIndex = 0;
        try {
          if (window.frameElement) {
            var allFrames = window.parent.document.querySelectorAll('iframe');
            var thisFrame = window.frameElement;
            var count = 0;
            for (var f = 0; f < allFrames.length; f++) {
              if (allFrames[f] === thisFrame) {
                currentIndex = count;
                break;
              }
              try {
                if (allFrames[f].contentDocument && allFrames[f].contentDocument.getElementById('ac-root')) {
                  count++;
                }
              } catch(e) {
                count++;
              }
            }
          }
        } catch(e) {
          currentIndex = 0;
        }

        var targetIndex = Math.min(currentIndex, blocks.length - 1);
        var data = parseBlock(blocks[targetIndex]);
        document.getElementById('ac-root').innerHTML = renderPanel(data);

      } catch (error) {
        document.getElementById('ac-root').innerHTML = '<div class="ac-loading">❌ ' + error.message + '</div>';
      }
    }

    $(function() { init(); });
  </script>
