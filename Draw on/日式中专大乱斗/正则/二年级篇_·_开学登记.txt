  function getLoreForRelation(charName, relVal) {
    var isOrphan = (val('fIdentity') || '').indexOf('孤儿') !== -1;

    var charDB = RELATION_LORE_DB[charName] || {};

      var isLocal = val('fOriginType') === '日本本土';
