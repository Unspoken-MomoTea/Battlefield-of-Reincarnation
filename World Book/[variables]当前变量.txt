      if (copy.道具) {

            ...item,
            状态: `${item.状态} // ${statusMark}`
          };
        }, {});

    }
    
    // 清理空对象，保持面板极致清爽
    if (_.isEmpty(copy.装备)) delete copy.装备;
    if (_.isEmpty(copy.道具)) delete copy.道具;

const stripHarvestFields = (seq) => _.omit(seq, ['下次产出日期', '下次产出游天']);

      if (tacticalAsset.建设序列) {

// 【核心修改】统一递归剔除「真属性」等后台结算快照键，严禁下发 AI，杜绝数据污染
const _stripForbidden = (root, forbiddenKeys = ['真属性']) => {
  const cloned = _.cloneDeep(root);
  const walk = (v) => {
    if (Array.isArray(v)) { v.forEach(walk); return; }
    if (v && typeof v === 'object') {
      for (const k of forbiddenKeys) delete v[k];
      // 仅隐藏实体顶层的强化计数，避免误删效果表中恰好名为“强化”的正常词条。
      if (Object.prototype.hasOwnProperty.call(v, '强化')
          && Object.prototype.hasOwnProperty.call(v, '效果')
          && (Object.prototype.hasOwnProperty.call(v, '品质') || Object.prototype.hasOwnProperty.call(v, '层级'))) {
        delete v.强化;
      }
      for (const k of Object.keys(v)) walk(v[k]);
    }
  };
  walk(cloned);
  return cloned;
};
