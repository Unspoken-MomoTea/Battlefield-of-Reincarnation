    <script>
      (function() {
        'use strict';
        var currentScript = document.currentScript;
        if (!currentScript) return;
        var wrapper = currentScript.closest('.bg3-wrapper');
        if (!wrapper) return;

        var rawBox = wrapper.querySelector('.bg3-raw-data');
        var capsule = wrapper.querySelector('.bg3-capsule');
        var detail = wrapper.querySelector('.bg3-detail');
        var capBadge = wrapper.querySelector('.cap-badge');
        var capMid = wrapper.querySelector('.cap-mid');
        var capTag = wrapper.querySelector('.cap-tag');

        if (!rawBox || !capsule || !detail) return;

        var rawText = (rawBox.textContent || '').trim();
        if (!rawText) return;

        function getQualityClass(quality) {
          if (!quality) return 'q-border-f';
          var q = quality.trim().toUpperCase();
          if (q === 'SSS' || q === 'Ⅸ') return 'q-border-sss';
          if (q === 'SS' || q === 'Ⅷ') return 'q-border-ss';
          if (q === 'S' || q === 'Ⅶ') return 'q-border-s';
          if (q === 'A' || q === 'Ⅵ') return 'q-border-a';
          if (q === 'B' || q === 'Ⅴ') return 'q-border-b';
          if (q === 'C' || q === 'Ⅳ') return 'q-border-c';
          if (q === 'D' || q === 'Ⅲ') return 'q-border-d';
          if (q === 'E' || q === 'Ⅱ') return 'q-border-e';
          return 'q-border-f';
        }

        function getRatingClass(ratingText) {
          if (!ratingText) return 'rating-success';
          if (ratingText.indexOf('大成功') !== -1) return 'rating-crit-success';
          if (ratingText.indexOf('大失败') !== -1) return 'rating-crit-failure';
          if (ratingText.indexOf('成功') !== -1 || ratingText.indexOf('胜出') !== -1) return 'rating-success';
          if (ratingText.indexOf('失败') !== -1 || ratingText.indexOf('落败') !== -1) return 'rating-failure';
          return 'rating-success';
        }

        function createDiceBox(val, label) {
          var slot = document.createElement('div');
          slot.className = 'dice-slot';

          var box = document.createElement('div');
          box.className = 'dice-box';

          var valNum = parseInt(val, 10);
          if (!isNaN(valNum)) {
            if (valNum >= 96) box.classList.add('crit-success');
            else if (valNum <= 5) box.classList.add('crit-failure');
          }

          var valElem = document.createElement('div');
          valElem.className = 'dice-value';
          valElem.textContent = isNaN(valNum) ? '--' : String(valNum);
          box.appendChild(valElem);

          var lblElem = document.createElement('div');
          lblElem.className = 'dice-label';
          lblElem.textContent = label || 'D100';

          slot.appendChild(box);
          slot.appendChild(lblElem);
          return slot;
        }

        function buildItemCard(item, category) {
          var card = document.createElement('div');
          card.className = 'item-card';

          var bar = document.createElement('div');
          bar.className = 'item-bar';

          var left = document.createElement('div');
          left.className = 'item-bar-left';

          var qBadge = document.createElement('span');
          qBadge.className = 'item-quality-badge ' + getQualityClass(item.quality);
          qBadge.textContent = item.quality || 'F';
          left.appendChild(qBadge);

          var nameSpan = document.createElement('span');
          nameSpan.className = 'item-name';
          nameSpan.textContent = item.name;
          left.appendChild(nameSpan);

          var right = document.createElement('div');
          right.className = 'item-bar-right';

          var snippetSpan = document.createElement('span');
          snippetSpan.className = 'item-snippet';
          if (item.shift) {
            snippetSpan.textContent = item.shift;
          } else if (category === 'item') {
            snippetSpan.textContent = item.qty ? ('× ' + item.qty) : (item.type !== '--' ? item.type : '');
          } else if (category === 'buff') {
            snippetSpan.textContent = item.duration || (item.type !== '--' ? item.type : '');
          } else if (category === 'asset') {
            snippetSpan.textContent = item.status || (item.scale ? ('LV' + item.scale) : '');
          } else {
            snippetSpan.textContent = item.rawAttr || item.cost || (item.type !== '--' ? item.type : '');
          }
          right.appendChild(snippetSpan);

          var hasDrawer = (item.effects && item.effects.length > 0) || (item.nestedSkills && item.nestedSkills.length > 0) || !!item.desc || (item.tags && item.tags.length > 0) || !!item.coreFunc;

          if (hasDrawer) {
            card.classList.add('has-drawer');
            var arrow = document.createElement('span');
            arrow.className = 'item-arrow';
            arrow.textContent = '▸';
            right.appendChild(arrow);
          }

              continue;
            }

            // 单品内部属性补充行
            if (curItem) {
              if (line.indexOf('| 原始属性:') === 0) {
                inMode = 'none';
                curItem.rawAttr = line.replace(/\|\s*原始属性:\s*/, '').replace(/\|/g, '').trim();
                continue;
              }
              if (line.indexOf('| 消耗:') === 0) {
                inMode = 'none';
                curItem.cost = line.replace(/\|\s*消耗:\s*/, '').replace(/\|/g, '').trim();
                continue;
              }
              if (line.indexOf('| 状态:') === 0) {
                inMode = 'none';
                curItem.status = line.replace(/\|\s*状态:\s*/, '').replace(/\|/g, '').trim();
                continue;
              }
              if (line.indexOf('| 核心功能:') === 0) {
                inMode = 'none';
                curItem.coreFunc = line.replace(/\|\s*核心功能:\s*/, '').replace(/\|/g, '').trim();
                continue;
              }
              if (line.indexOf('| 标签:') === 0) {
                inMode = 'none';
                var tRaw = line.replace(/\|\s*标签:\s*/, '').replace(/\|/g, '').trim();
                var tMatches = tRaw.match(/\[(.*?)\]/g);
                if (tMatches) {
                  for (var tmI = 0; tmI < tMatches.length; tmI++) {
                    curItem.tags.push(tMatches[tmI].replace(/[\[\]]/g, '').trim());
                  }
                } else if (tRaw) { curItem.tags.push(tRaw); }
                continue;
              }

              if (line.indexOf('| 效果:') === 0) {
                inMode = 'eff';
                var effIn = line.replace(/\|\s*效果:\s*/, '').replace(/\|/g, '').trim();
                if (effIn) curItem.effects.push(effIn);
                continue;
              }

              if (line.indexOf('| 技能:') === 0) {
                inMode = 'skill';
                var skIn = line.replace(/\|\s*技能:\s*/, '').replace(/\|/g, '').trim();
                if (skIn) curItem.nestedSkills.push(skIn);
                continue;
              }

              if (line.indexOf('| 描述:') === 0) {
                inMode = 'desc';
                var descIn = line.replace(/\|\s*描述:\s*/, '').replace(/\|/g, '').trim();
                if (descIn) curItem.desc = descIn;
                continue;
              }

              // 处于多行模式时的后续行收集
              var cleanLine = line.replace(/^\|\s*/, '').replace(/\s*\|$/, '').trim();
              if (cleanLine) {
                if (inMode === 'eff') curItem.effects.push(cleanLine);
                else if (inMode === 'skill') curItem.nestedSkills.push(cleanLine);
                else if (inMode === 'desc') curItem.desc += (curItem.desc ? ' ' : '') + cleanLine;
              }
            }
          }

          var firstProfile = null;
          for (var s = 0; s < sectionsOrder.length; s++) {
            if (sectionsOrder[s].profile) {
              firstProfile = sectionsOrder[s].profile;
              break;
            }
          }

          if (!eMeta.type) {
            if (firstProfile) {
              eMeta.type = '实体登场';
            } else if (sectionsOrder.some(function(sec) { return sec.tag === 'skill' || sec.tag === 'form' || sec.tag === 'buff' || sec.tag === 'relation'; })) {
              eMeta.type = '状态变迁';
            } else {
              eMeta.type = '物品获取';
            }
          }

          if (eMeta.type === '实体登场' || firstProfile) {
            capBadge.textContent = '👤 实体侦测';
            var entName = firstProfile ? firstProfile.name : '--';
            capMid.textContent = eMeta.source ? (entName + ' · ' + eMeta.source) : entName;
            capTag.className = 'cap-tag ' + (firstProfile ? getQualityClass(firstProfile.tier) : 'q-border-f');
            capTag.textContent = firstProfile ? ('阶位: ' + firstProfile.tier) : '新实体';
          } else if (eMeta.type === '状态变迁' || eMeta.type === '形态变动') {
            capBadge.textContent = '🏷️ 变迁与突破';
            capMid.textContent = eMeta.source || '状态与能力变迁';
            capTag.className = 'cap-tag q-border-d';
            var shiftCounts = [];
            sectionsOrder.forEach(function(sec) {
              if (sec.tag === 'skill' && sec.items.length) shiftCounts.push(sec.items.length + ' 技能');
              if (sec.tag === 'form' && sec.items.length) shiftCounts.push(sec.items.length + ' 形态');
              if (sec.tag === 'buff' && sec.items.length) shiftCounts.push(sec.items.length + ' 状态');
              if (sec.tag === 'relation' && sec.relations.length) shiftCounts.push(sec.relations.length + ' 记录');
              if (sec.tag === 'item' && sec.items.length) shiftCounts.push(sec.items.length + ' 道具');
            });
            capTag.textContent = shiftCounts.length ? shiftCounts.join(' · ') : (eMeta.type || '能力变迁');
          } else {
            capBadge.textContent = '❖ 物品获取';
            capMid.textContent = eMeta.source || '收纳整理';
            capTag.className = 'cap-tag q-border-e';
            var lootCounts = [];
            sectionsOrder.forEach(function(sec) {
              if (sec.tag === 'equip' && sec.items.length) lootCounts.push(sec.items.length + ' 装备');
              if (sec.tag === 'item' && sec.items.length) lootCounts.push(sec.items.length + ' 道具');
              if (sec.tag === 'asset' && sec.items.length) lootCounts.push(sec.items.length + ' 资产');
            });
            capTag.textContent = lootCounts.length ? lootCounts.join(' · ') : '已收纳';
          }

          detail.innerHTML = '';

          var hDiv2 = document.createElement('div');
          hDiv2.className = 'detail-header';
          var tMainText = (eMeta.type === '实体登场' || firstProfile) ? '❖ 实体详细档案 ❖' : ((eMeta.type === '状态变迁' || eMeta.type === '形态变动') ? '❖ 能力突破与变迁 ❖' : '❖ 物品收纳清单 ❖');
          var tMainEl = document.createElement('div');
          tMainEl.className = 'detail-title-main';
          tMainEl.textContent = tMainText;
          hDiv2.appendChild(tMainEl);

          if (eMeta.source) {
            var tSubEl = document.createElement('div');
            tSubEl.className = 'detail-title-sub';
            tSubEl.textContent = '来源: ' + eMeta.source;
            hDiv2.appendChild(tSubEl);
          }
          detail.appendChild(hDiv2);

          // 核心渲染：严格按照 sectionsOrder 顺序依次动态输出到 DOM 中
          for (var k = 0; k < sectionsOrder.length; k++) {
            var secObj = sectionsOrder[k];

            if (secObj.tag === 'profile' && secObj.profile) {
              var prof = secObj.profile;
              var pCard = document.createElement('div');
              pCard.className = 'entity-profile-card';

              var pGrid = document.createElement('div');
              pGrid.className = 'entity-grid';

              function addPField(lbl, val) {
                if (!val) return;
                var fEl = document.createElement('div');
                fEl.className = 'entity-grid-field';
                fEl.textContent = lbl + ': ';
                var fSpan = document.createElement('span');
                fSpan.textContent = val;
                fEl.appendChild(fSpan);
                pGrid.appendChild(fEl);
              }

              addPField('实体', prof.name);
              addPField('层级', prof.tier);
              addPField('好感度', prof.favor);
              addPField('种族', prof.race);
              addPField('职业', prof.occupation);
              if (prof.count) addPField('群体数量', prof.count);
              pCard.appendChild(pGrid);

              function addPExtra(lbl, val) {
                if (!val) return;
                var exDiv = document.createElement('div');
                exDiv.className = 'entity-extra';
                var exLbl = document.createElement('span');
                exLbl.className = 'entity-extra-label';
                exLbl.textContent = lbl + '：';
                exDiv.appendChild(exLbl);
                exDiv.appendChild(document.createTextNode(val));
                pCard.appendChild(exDiv);
              }

              addPExtra('观测机制', prof.mechanism);
              addPExtra('真实态度', prof.attitude);
              addPExtra('外貌着装', prof.appearance);
              addPExtra('身份风味', prof.desc);

              detail.appendChild(pCard);

            } else if (secObj.tag === 'relation' && secObj.relations && secObj.relations.length > 0) {
              var rWrap = document.createElement('div');
              rWrap.className = 'section-wrap';
              var rTitle = document.createElement('div');
              rTitle.className = 'section-title';
              rTitle.textContent = secObj.title;
              rWrap.appendChild(rTitle);

              var rList = document.createElement('div');
              rList.className = 'items-col';
              for (var rl = 0; rl < secObj.relations.length; rl++) {
                var rItem = secObj.relations[rl];
                var rc = document.createElement('div');
                rc.className = 'relation-card';

                var rTop = document.createElement('div');
                rTop.className = 'relation-top';
                var rName = document.createElement('span');
                rName.className = 'relation-name';
                rName.textContent = rItem.name;
                var rShift = document.createElement('span');
                rShift.className = 'relation-shift';
                rShift.textContent = rItem.shift;
                rTop.appendChild(rName);
                rTop.appendChild(rShift);

                var rBody = document.createElement('div');
                rBody.className = 'relation-body';
                if (rItem.attitude) {
                  var strongAtt = document.createElement('strong');
                  strongAtt.textContent = '当前状态/态度：';
                  rBody.appendChild(strongAtt);
                  rBody.appendChild(document.createTextNode(rItem.attitude));
                }

                if (rItem.reason) {
                  if (rItem.attitude) rBody.appendChild(document.createElement('br'));
                  var strongRea = document.createElement('strong');
                  strongRea.textContent = '变动成因：';
                  rBody.appendChild(strongRea);
                  rBody.appendChild(document.createTextNode(rItem.reason));
                }

                rc.appendChild(rTop);
                rc.appendChild(rBody);
                rList.appendChild(rc);
              }
              rWrap.appendChild(rList);
              detail.appendChild(rWrap);

            } else if (secObj.items && secObj.items.length > 0) {
              var sWrap = document.createElement('div');
              sWrap.className = 'section-wrap';
              var sTitle = document.createElement('div');
              sTitle.className = 'section-title';
              sTitle.textContent = secObj.title;
              sWrap.appendChild(sTitle);

              var sList = document.createElement('div');
              sList.className = 'items-col';
              for (var m = 0; m < secObj.items.length; m++) {
                sList.appendChild(buildItemCard(secObj.items[m], secObj.tag));
              }
              sWrap.appendChild(sList);
              detail.appendChild(sWrap);
            }
          }

          if (eMeta.consequence) {
            var cFoot = document.createElement('div');
            cFoot.className = 'consequence-card';
            var cTitle2 = document.createElement('div');
            cTitle2.className = 'consequence-title';
            cTitle2.textContent = '现实因果变动';
            var cBody2 = document.createElement('div');
            cBody2.className = 'consequence-body';
            cBody2.textContent = eMeta.consequence;
            cFoot.appendChild(cTitle2);
            cFoot.appendChild(cBody2);
            detail.appendChild(cFoot);
          }

          if (eMeta.status) {
            var sFoot = document.createElement('div');
            sFoot.className = 'consequence-card';
            var sTitle2 = document.createElement('div');
            sTitle2.className = 'consequence-title';
            sTitle2.textContent = '最终状态更新';
            var sBody2 = document.createElement('div');
            sBody2.className = 'consequence-body';
            sBody2.textContent = eMeta.status;
            sFoot.appendChild(sTitle2);
            sFoot.appendChild(sBody2);
            detail.appendChild(sFoot);
          }
        }

        capsule.addEventListener('click', function() {
          var isOpen = detail.classList.contains('open');
          if (isOpen) {
            detail.classList.remove('open');
            capsule.classList.remove('active');
          } else {
            detail.classList.add('open');
            capsule.classList.add('active');
          }
        });

      })();
    </script>
  </div>

</body>
</html>
