<script>
(function() {
  var currentWorldbook = null;
  var dlcData = {};

  /* 解析正则：匹配 [DLC][角色][项目名称] 或 [DLC][扩展][项目名称] */
  var dlcRegex = /^\[DLC\]\[(?:角色|扩展)\]\[([^\]]+)\].*$/;

  /* 初始化获取世界书名称 */
  async function initWorldbookName() {
    try {
      if (typeof getCharWorldbookNames === 'function') {
        currentWorldbook = getCharWorldbookNames('current').primary;
      }
      return !!currentWorldbook;
    } catch (e) {
      console.error('[DLC终端] 获取世界书名称失败:', e);
      return false;
    }
  }

  /* 加载并解析DLC数据 */
  window.loadDLCs = async function() {
    var container = document.getElementById('dlc-list-container');
    container.innerHTML = '<div class="dlc-loading"><div class="dlc-spinner"></div>正在扫描 DLC 模块...</div>';

    if (!currentWorldbook) {
      var ok = await initWorldbookName();
      if (!ok) {
        container.innerHTML = '<div class="dlc-empty">未检测到绑定的世界书<br>请先为当前角色绑定世界书</div>';
        return;
      }
    }

    try {
      if (typeof getWorldbook !== 'function') throw new Error('酒馆版本不支持 getWorldbook API');

      var entries = await getWorldbook(currentWorldbook);
      dlcData = {};

      entries.forEach(function(entry) {
        if (!entry.name) return;
        var match = entry.name.match(dlcRegex);
        if (match) {
          var projectName = match[1];
          if (!dlcData[projectName]) {
            dlcData[projectName] = { total: 0, enabled: 0, uids: [] };
          }
          dlcData[projectName].total += 1;
          dlcData[projectName].uids.push(entry.uid);
          if (entry.enabled) {
            dlcData[projectName].enabled += 1;
          }
        }
      });

      renderDLCList();

    } catch (error) {
      console.error('[DLC终端] 读取失败:', error);
      container.innerHTML = '<div class="dlc-empty">读取失败: ' + error.message + '</div>';
      dlcToast('读取世界书失败', 'error');
    }
  };

  /* 渲染列表 */
  function renderDLCList() {
    var container = document.getElementById('dlc-list-container');
    var projects = Object.keys(dlcData);

    if (projects.length === 0) {
      container.innerHTML = '<div class="dlc-empty">未扫描到任何有效的 DLC 条目<br><span style="font-size:10px;opacity:0.5;display:block;margin-top:6px;">条目命名格式：[DLC][角色][名称]xxx</span></div>';
      return;
    }

    var html = '';
    projects.forEach(function(project) {
      var data = dlcData[project];
      var isChecked = data.enabled > 0;
      var statusText = '';
      if (data.enabled === data.total) {
        statusText = '已全开';
      } else if (data.enabled === 0) {
        statusText = '已关闭';
      } else {
        statusText = '部分开启 (' + data.enabled + '/' + data.total + ')';
      }

      html += '<div class="dlc-item">'
        + '<div class="dlc-item-info">'
        + '<div class="dlc-item-name">' + project + ' <span class="dlc-item-badge">DLC</span></div>'
        + '<div class="dlc-item-stats">' + data.total + ' 个条目 · <span id="status-' + project + '">' + statusText + '</span></div>'
        + '</div>'
        + '<label class="switch">'
        + '<input type="checkbox" id="toggle-' + project + '" ' + (isChecked ? 'checked' : '') + ' onchange="toggleDLC(\'' + project + '\', this.checked)">'
        + '<span class="slider"></span>'
        + '</label>'
        + '</div>';
    });

    container.innerHTML = html;
  }

  /* 切换DLC开关状态 */
  window.toggleDLC = async function(project, turnOn) {
    var checkbox = document.getElementById('toggle-' + project);
    var statusEl = document.getElementById('status-' + project);
    checkbox.disabled = true;

    try {
      var entries = await getWorldbook(currentWorldbook);
      var targetUids = dlcData[project].uids;
      var updateCount = 0;

      var updatedEntries = entries.map(function(entry) {
        if (targetUids.indexOf(entry.uid) !== -1) {
          updateCount++;
          var copy = {};
          for (var k in entry) {
            if (Object.prototype.hasOwnProperty.call(entry, k)) copy[k] = entry[k];
          }
          copy.enabled = turnOn;
          return copy;
        }
        return entry;
      });

      await replaceWorldbook(currentWorldbook, updatedEntries);

      dlcData[project].enabled = turnOn ? dlcData[project].total : 0;
      statusEl.textContent = turnOn ? '已全开' : '已关闭';

      dlcToast('[' + project + '] ' + (turnOn ? '已开启' : '已关闭') + ' (' + updateCount + '个条目)', 'success');

    } catch (error) {
      console.error('[DLC终端] 切换失败:', error);
      checkbox.checked = !turnOn;
      dlcToast('操作失败，请重试', 'error');
    } finally {
      checkbox.disabled = false;
    }
  };

  /* 返回封面页 */
  window.dlcBackCover = async function() {
    try {
      if (typeof getChatMessages !== 'function' || typeof setChatMessage !== 'function') {
        dlcToast('当前环境不支持返回封面', 'error');
        return;
      }
      var messages = await getChatMessages('0', { include_swipe: true });
      if (messages && messages[0] && messages[0].swipes && messages[0].swipes[0]) {
        await setChatMessage(messages[0].swipes[0], 0, {
          swipe_id: 0,
          refresh: 'display_and_render_current'
        });
      } else {
        dlcToast('未找到封面页数据', 'error');
      }
    } catch (error) {
      console.error('[DLC终端] 返回封面失败:', error);
      dlcToast('返回封面失败', 'error');
    }
  };

  /* Toast通知 */
  function dlcToast(msg, type) {
    if (typeof toastr !== 'undefined') {
      if (type === 'success') toastr.success(msg);
      else if (type === 'error') toastr.error(msg);
      else toastr.info(msg);
    } else {
      console.log('[' + (type || 'info') + '] ' + msg);
    }
  }

  /* 延迟加载 */
  setTimeout(function() { loadDLCs(); }, 500);
})();
</script>

</body>
</html>
