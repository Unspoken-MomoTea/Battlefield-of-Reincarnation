    var currentFilter = '全部';

    function getV(path, def) {
      return _.get(getAllVariables(), 'stat_data.' + path, def);

    function formatTime(raw) {
      if (!raw || raw === 'N/A') return '--';
      var parts = raw.split('-');
      if (parts.length >= 5) return parts[1] + parts[2] + ' ' + parts[3] + ' ' + parts[4];
      if (parts.length >= 3) return parts[1] + parts[2];
      return raw;
    }

    function gradeClass(g) {
      return 'g-' + (g || 'e').charAt(0).toLowerCase();
    }

    /* 渲染效果状态胶囊列表 */
    function renderEffects(effects, uniquePrefix) {
      if (!effects || Object.keys(effects).length === 0) return '<div class="s-empty">无状态</div>';
      var html = '<div class="s-effects-container">';
      var prefix = uniquePrefix || 'fx';

      Object.entries(effects).forEach(function(e, idx) {
        var name = e[0], d = e[1] || {};
        var type = d['类型'] || '特殊';
        var drawerId = prefix + '-drawer-' + idx;

        html += '<div class="s-effect-capsule" data-drawer="' + drawerId + '">';
        html += '<span class="s-capsule-dot ct-' + type + '"></span>';
        html += '<span class="s-capsule-name">' + name + '</span>';
        html += '</div>';

        html += '<div class="s-capsule-drawer ct-' + type + '" id="' + drawerId + '">';
        html += '<div class="s-drawer-row"><span>分类：</span>' + type + '</div>';
        if (d['效果']) html += '<div class="s-drawer-row"><span>效果：</span>' + d['效果'] + '</div>';
        html += '<div class="s-drawer-row"><span>剩余：</span>' + (d['剩余时间'] || '永久') + '</div>';
        if (d['来源']) html += '<div class="s-drawer-row"><span>来源：</span>' + d['来源'] + '</div>';
        html += '</div>';
      });

      html += '</div>';
      return html;
    }

    function renderAbilityGrid(ab) {
      if (!ab) return '';
      var names = ['学力','身体能力','灵活思考力','社会贡献性'];
      var html = '<div class="s-ability-grid">';
      names.forEach(function(name) {
        var d = ab[name] || {};
        var grade = d['评级'] || '-';
        var score = d['分数'] || 0;
        var exp = d['经验值'] || 0;
        html += '<div class="s-ab-item">';
        html += '<div class="s-ab-name">' + name + '</div>';
        html += '<div class="s-ab-grade ' + gradeClass(grade) + '">' + grade + '</div>';
        html += '<div class="s-ab-score">' + score + '/100</div>';
        html += '<div class="s-ab-bar"><div class="s-ab-fill" style="width:' + exp + '%"></div></div>';
        html += '</div>';
      });
      html += '</div>';
      return html;
    }

    function populateData() {
      var v = getAllVariables();

      $('#g-time').text(formatTime(_.get(v, 'stat_data.学生证.当前时间', '')));
      $('#g-loc').text(_.get(v, 'stat_data.学生证.当前地点', '--'));

      $('#stu-name').text(_.get(v, 'stat_data.学生证.姓名', '--'));

      var pp = _.get(v, 'stat_data.学生证.个人点数', 0);
      $('#stu-pp').text(Number(pp).toLocaleString() + ' PP');

      var abilities = ['学力','身体能力','灵活思考力','社会贡献性'];
      abilities.forEach(function(name) {

        var $el = $('.s-ab-item[data-ab="' + name + '"]');
        var grade = d['评级'] || '-';
        $el.find('.s-ab-grade').text(grade).attr('class', 's-ab-grade ' + gradeClass(grade));
        $el.find('.s-ab-score').text((d['分数'] || 0) + '/100');
        $el.find('.s-ab-fill').css('width', (d['经验值'] || 0) + '%');
      });

      var effects = _.get(v, 'stat_data.学生证.效果状态', {});
      $('#stu-effects').html(renderEffects(effects, 'user'));

      var classRank = _.get(v, 'stat_data.班级终端.班级排名', {});
      var sorted = Object.entries(classRank).sort(function(a, b) {
        var ra = parseInt(((a[1]['排名'] || '第4名').match(/\d+/) || ['4'])[0]);
        var rb = parseInt(((b[1]['排名'] || '第4名').match(/\d+/) || ['4'])[0]);
        return ra - rb;

      var cHtml = '';

    function renderNPCList() {
      var v = getAllVariables();
      var maleNPCs = _.get(v, 'stat_data.学生系统.男', {});
      var femaleNPCs = _.get(v, 'stat_data.学生系统.女', {});

      function npcListHtml(npcs, gender) {
        if (!npcs || Object.keys(npcs).length === 0) return '';
        var html = '';
        Object.entries(npcs).forEach(function(e) {
          var name = e[0], d = e[1] || {};

          var classKey = d['班级'] || '';
          if (currentFilter !== '全部') {
            if (currentFilter === '其他') {
              if (['A班', 'B班', 'C班', 'D班'].includes(classKey)) return;
            } else {
              if (classKey !== currentFilter) return;
            }
          }

          var present = d['在场状态'] === '在场';
          var att = d['态度'] || '陌生';
          var cls = d['实际班级'] || '';
          html += '<div class="s-npc-item" data-npc="' + name + '" data-gender="' + gender + '">'
            + '<span class="s-npc-dot ' + (present ? 'in' : 'out') + '"></span>'
            + '<span class="s-npc-name">' + name + '</span>'
            + '<span class="s-npc-class">' + cls + '</span>'
            + '<span class="s-npc-att att-' + att + '">' + att + '</span>'
            + '</div>';
        });
        return html;
      }

      var fHtml = npcListHtml(femaleNPCs, '女');
      var mHtml = npcListHtml(maleNPCs, '男');

      var nHtml = '';
      if (fHtml) nHtml += '<div class="s-npc-section">♀ 女性角色</div>' + fHtml;
      if (mHtml) nHtml += '<div class="s-npc-section">♂ 男性角色</div>' + mHtml;
      $('#npc-list').html(nHtml || '<div class="s-empty">暂无符合条件的追踪角色</div>');
    }

    async function init() {
      await waitGlobalInitialized('Mvu');
      populateData();

      eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, function() {
        populateData();
      });

      /* 读取主题设置 */
      var savedTheme = localStorage.getItem('cote_status_theme') || 'dark';
      $('.s-container').attr('data-theme', savedTheme);
      $('.s-theme-card').removeClass('active');
      $('.s-theme-card[data-theme-id="' + savedTheme + '"]').addClass('active');

      $('.s-tab-btn').on('click', function() {
        var target = $(this).data('tab');
        $('.s-tab-btn').removeClass('active');
        $(this).addClass('active');
        $('.s-panel').removeClass('active');
        $('#panel-' + target).addClass('active');
        if (target === 'sys') { closeNpcDetail(); }
      });

      /* 打开个性化主题弹窗 */
      $('.s-theme-trigger-btn').on('click', function() {
        $('.s-theme-modal').fadeIn(200);
      });

      /* 关闭个性化主题弹窗 */
      $('.s-modal-close-btn').on('click', function() {
        $('.s-theme-modal').fadeOut(150);
      });

      /* 在弹窗中点击卡片，切换并应用主题 */
      $('.s-theme-card').on('click', function() {
        var t = $(this).data('theme-id');
        $('.s-container').attr('data-theme', t);
        $('.s-theme-card').removeClass('active');
        $(this).addClass('active');
        localStorage.setItem('cote_status_theme', t);
      });

      $(document).on('click', '.s-filter-btn', function() {
        $('.s-filter-btn').removeClass('active');
        $(this).addClass('active');
        currentFilter = $(this).data('filter');
        renderNPCList();
      });

      $(document).on('click', '.s-npc-item', function() {
        var name = $(this).data('npc');
        var gender = $(this).data('gender');
        showNpcDetail(name, gender);
      });

      $(document).on('click', '.s-detail-back', function() {
        closeNpcDetail();
      });

      $(document).on('click', '.s-card-toggle', function() {
        var target = $(this).data('target');
        var body = $('#' + target);
        var self = $(this);

        body.slideToggle(200, function() {
          self.toggleClass('open', body.is(':visible'));
        });
      });

      /* 效果状态胶囊标签抽屉展开 */
      $(document).on('click', '.s-effect-capsule', function(e) {
        e.stopPropagation();
        var drawerId = $(this).data('drawer');
        var drawer = $('#' + drawerId);

        $(this).toggleClass('active');
        drawer.slideToggle(150);
      });
    }

    $(errorCatched(init));
